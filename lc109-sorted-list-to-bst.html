Algorithm

private ListNode head;

public TreeNode sortedListToBST(ListNode head) {
    this.head = head;
    int n = 0;
    for (ListNode p = head; p != null; p = p.next) {
        n++;
    }
    return build(0, n - 1);
}

private TreeNode build(int lo, int hi) {
    if (lo > hi) {
        return null;
    }
    int mid = (lo + hi) / 2;
    TreeNode left  = build(lo, mid - 1);      // walk left subtree first (inorder)
    TreeNode root  = new TreeNode(head.val);  // head is at the mid-th node
    head = head.next;                       // advance head past it
    root.left  = left;
    root.right = build(mid + 1, hi);           // then right subtree
    return root;
}
