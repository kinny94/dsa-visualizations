LC 92 · Reverse Linked List II · Head-Insertion Surgery

head = [1, 2, 3, 4, 5], left = 2, right = 4 · each iter yanks curr's successor and plugs it right after prev
current code line prev (fixed anchor) curr (sinks to tail of section) next (gets pulled to front)

Algorithm

public ListNode reverseBetween(ListNode head, int left, int right) {
    if (head == null || left == right) { return head; }
    ListNode dummy = new ListNode(0, head);
    ListNode prev = dummy;
    for (int i = 1; i < left; i++) { prev = prev.next; }
    ListNode curr = prev.next;
    for (int i = 0; i < right - left; i++) {
        ListNode next = curr.next;
        curr.next = next.next;     // detach next from after curr
        next.next = prev.next;     // stitch next → old front
        prev.next = next;          // make next the new front
    }
    return dummy.next;
}
