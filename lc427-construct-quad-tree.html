public Node construct(int[][] grid) {
    return build(grid, 0, 0, grid.length);
}

private Node build(int[][] g, int row, int col, int size) {
    if (isUniform(g, row, col, size)) {
        return new Node(g[row][col] == 1, true);
    }
    int half = size / 2;
    Node tl = build(g, row,        col,        half);
    Node tr = build(g, row,        col + half, half);
    Node bl = build(g, row + half, col,        half);
    Node br = build(g, row + half, col + half, half);
    return new Node(true, false, tl, tr, bl, br);
}

private boolean isUniform(int[][] g, int r, int c, int s) {
    int first = g[r][c];
    for (int i = r; i < r + s; i++) {
        for (int j = c; j < c + s; j++) {
            if (g[i][j] != first) { return false; }
        }
    }
    return true;
}
