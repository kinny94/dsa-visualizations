class MovingAverage {
    private Deque<Integer> q;
    private int size;
    private double sum;

    public MovingAverage(int size) {
        this.q = new ArrayDeque<>();
        this.size = size;
        this.sum = 0;
    }

    public double next(int val) {
        if (q.size() == size) {
            sum -= q.poll();          // evict oldest
        }
        q.offer(val);                 // enqueue new
        sum += val;
        return sum / q.size();
    }
}
